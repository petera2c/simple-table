export class NativeTemplatesDemoComponent {
  @Input() height: string | number = "360px";
  @Input() theme?: Theme;

  readonly columns: AngularColumnDef<DirectoryPerson>[] = directoryColumns;
  readonly people = signal<DirectoryPerson[]>(DIRECTORY_SEED.map((row) => ({ ...row })));
  readonly sortLabel = signal("None");
  readonly selectedLabel = signal("None");
  readonly lastPing = signal<string | null>(null);

  private readonly selectedIds = new Set<number>();
  private nextId = DIRECTORY_SEED.length + 1;

  getRowId = ({ row }: GetRowIdParams<DirectoryPerson>) => row.id;

  statusStyle(value: unknown) {
    const status = String(value) as PersonStatus;
    return STATUS_STYLE[status] ?? STATUS_STYLE.offline;
  }

  initials(name: string): string {
    return name
      .split(" ")
      .slice(0, 2)
      .map((part) => part[0] ?? "")
      .join("")
      .toUpperCase();
  }

  onSort(sort: SortColumn | null): void {
    if (!sort) {
      this.sortLabel.set("None");
      return;
    }
    this.sortLabel.set(`${String(sort.key.accessor)} ${sort.direction}`);
  }

  onSelect(event: RowSelectionChangeProps<DirectoryPerson>): void {
    if (event.isSelected) {
      this.selectedIds.add(event.row.id);
    } else {
      this.selectedIds.delete(event.row.id);
    }
    const names = this.people()
      .filter((row) => this.selectedIds.has(row.id))
      .map((row) => row.name);
    this.selectedLabel.set(names.length ? names.join(", ") : "None");
  }

  ping(row: DirectoryPerson): void {
    this.lastPing.set(row.name);
  }

  cycleStatus(row: DirectoryPerson): void {
    const index = STATUS_CYCLE.indexOf(row.status);
    const next = STATUS_CYCLE[(index + 1) % STATUS_CYCLE.length];
    this.people.update((rows) =>
      rows.map((person) => (person.id === row.id ? { ...person, status: next } : person)),
    );
  }

  addPerson(): void {
    const id = this.nextId++;
    this.people.update((rows) => [
      ...rows,
      {
        id,
        name: `New hire ${id}`,
        role: "Engineer",
        team: "Platform",
        status: "active",
      },
    ]);
  }

  clearPeople(): void {
    this.people.set([]);
    this.selectedIds.clear();
    this.selectedLabel.set("None");
    this.lastPing.set(null);
  }
}
